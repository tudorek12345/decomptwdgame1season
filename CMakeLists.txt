cmake_minimum_required(VERSION 3.24)
project(TWDSeasonOneRecovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(CTest)
set(TWD_CAPTURE_FIXTURES_AVAILABLE OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/original_geometry_bytes.hpp" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/original_hash_symbols.hpp")
  set(TWD_CAPTURE_FIXTURES_AVAILABLE ON)
endif()
option(TWD_BUILD_CAPTURE_TESTS "Build local x86 machine-code comparison harnesses (requires private fixtures)" ${TWD_CAPTURE_FIXTURES_AVAILABLE})
add_library(twd_reconstructed STATIC reconstructed/geometry.cpp reconstructed/geometry.hpp
  reconstructed/hash.cpp reconstructed/hash.hpp reconstructed/symbol.cpp reconstructed/symbol.hpp)
target_include_directories(twd_reconstructed PUBLIC reconstructed)
if(MSVC)
  target_compile_options(twd_reconstructed PRIVATE /W4 /fp:strict /Zi /Zo)
endif()
file(GLOB RECOVERED_NATIVE CONFIGURE_DEPENDS recovered/native/*.cpp recovered/native/*.h recovered/native-additional/*.cpp recovered/native-additional/*.h recovered/native-startup/*.cpp recovered/native-startup/*.h recovered/hash-symbols/*.cpp)
file(GLOB_RECURSE RECOVERED_LUA CONFIGURE_DEPENDS recovered/lua/episodes/*.lua)
add_custom_target(decompiled_reference SOURCES ${RECOVERED_NATIVE} ${RECOVERED_LUA})
if(RECOVERED_NATIVE OR RECOVERED_LUA)
  source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/recovered" PREFIX "Reference only" FILES ${RECOVERED_NATIVE} ${RECOVERED_LUA})
endif()
if(BUILD_TESTING)
  add_executable(source_contracts tests/source_contracts.cpp)
  target_link_libraries(source_contracts PRIVATE twd_reconstructed)
  add_test(NAME public_source_contracts COMMAND source_contracts)
endif()
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_library(twd_runtime_geometry SHARED reconstructed/runtime_geometry.cpp reconstructed/runtime_geometry.def
    reconstructed/runtime_hash_symbols.cpp reconstructed/telemetry.hpp)
  set_target_properties(twd_runtime_geometry PROPERTIES OUTPUT_NAME twd_reconstructed ARCHIVE_OUTPUT_NAME twd_runtime_geometry)
  target_link_libraries(twd_runtime_geometry PRIVATE twd_reconstructed)
  target_compile_options(twd_runtime_geometry PRIVATE /Zi /Zo)
  target_link_options(twd_runtime_geometry PRIVATE /DEBUG:FULL)
  if(TWD_BUILD_CAPTURE_TESTS AND BUILD_TESTING)
  if(NOT TWD_CAPTURE_FIXTURES_AVAILABLE)
    message(FATAL_ERROR "Capture comparison tests require locally generated tests/original_* headers. They are intentionally absent from the public repository.")
  endif()
  add_executable(geometry_differential tests/geometry_differential.cpp tests/original_geometry_bytes.hpp)
  target_link_libraries(geometry_differential PRIVATE twd_reconstructed)
  target_compile_options(geometry_differential PRIVATE /W4 /fp:strict)
  enable_testing()
  add_test(NAME captured_geometry_comparison COMMAND geometry_differential)
  add_test(NAME exported_dll_geometry_comparison COMMAND geometry_differential $<TARGET_FILE:twd_runtime_geometry>)
  add_executable(hash_symbol_differential tests/hash_symbol_differential.cpp tests/original_hash_symbols.hpp)
  target_link_libraries(hash_symbol_differential PRIVATE twd_reconstructed)
  target_compile_options(hash_symbol_differential PRIVATE /W4)
  add_test(NAME captured_hash_symbol_comparison COMMAND hash_symbol_differential)
  add_test(NAME exported_dll_hash_symbol_comparison COMMAND hash_symbol_differential $<TARGET_FILE:twd_runtime_geometry>)
  endif()
endif()
if(WIN32)
  add_executable(twd_image_rebuild tools/image_rebuild.cpp)
  target_link_libraries(twd_image_rebuild PRIVATE bcrypt)
  target_compile_options(twd_image_rebuild PRIVATE /W4)
endif()
